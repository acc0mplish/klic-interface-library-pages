import React,{useEffect,useMemo,useRef,useState} from 'react';
import {ChartShowcase,useKlicChartTone} from '../charts/ChartShowcase';
import {CandlestickChart,createCandles} from '../charts/bjork/candlestick-chart';
import {ForecastFan,createForecastSeries} from '../charts/bjork/forecast-fan';

type Variant='candlestick'|'forecast'|'drawdown'|'depth'|'orderbook'|'state-timeline'|'spark-table';
type Density='compact'|'normal'|'relaxed';
const START=Date.UTC(2025,0,2),HOUR=3600000;
const CANDLES=createCandles(73,260,{start:START,base:184.5,vol:.018});
const FORECAST=createForecastSeries(42,{weeks:78,horizon:30,base:248,start:Date.UTC(2024,0,1)});
const EQUITY=createEquityCurve(19,756,Date.UTC(2023,0,2));
const DOMAIN:[number,number]=[Date.UTC(2026,9,23),Date.UTC(2026,9,24)];
const LANES=createIncidentLanes(57,['API gateway','Payments','Event queue','Search','Warehouse','Auth service'],DOMAIN).map((lane,index)=>({...lane,label:['API 게이트웨이','결제','이벤트 큐','검색','데이터 웨어하우스','인증 서비스'][index]}));
const STATES:TimelineState[]=[{id:'ok',label:"정상 운영",tone:'ok'},{id:'degraded',label:"성능 저하",tone:'warn'},{id:'down',label:"장애",tone:'down'},{id:'maintenance',label:"점검",tone:'idle'}];
const SPARK_CONFIG=[["매출",'USD',248000,.004,22000,9000,'up'],["활성 사용자",'users',18200,.003,1800,500,'up'],["요청 수",'requests',940000,.002,88000,18000,'up'],["전환율",'percent',3.2,.001,.22,.1,'up'],["95백분위 지연 시간",'ms',185,-.001,12,7,'down'],["오류율",'percent',.72,-.002,.08,.035,'down'],["요청당 비용",'USD',.018,-.001,.001,.0003,'down']] as const;
const fmtDate=(t:number)=>new Intl.DateTimeFormat('ko-KR',{month:'short',day:'numeric',timeZone:'UTC'}).format(t);
const ctl={display:'flex',alignItems:'center',gap:8,fontSize:12} as const;
const group={display:'flex',alignItems:'center',gap:18,flexWrap:'wrap'} as const;
const formatMoney=(v:number)=>'$'+v.toLocaleString('ko-KR',{maximumFractionDigits:2});
function SparkDemo({tone,density}:{tone:'light'|'dark';density:Density}){
 const [period,setPeriod]=useState(90),[index,setIndex]=useState<number|null>(null),[selected,setSelected]=useState<string|null>('revenue');
 const rows=useMemo<SparkRow[]>(()=>SPARK_CONFIG.map(([label,unit,base,trend,weekly,noise,goodDirection],i)=>({id:['revenue','active-users','requests','conversion','p95-latency','error-rate','cost-/-request'][i],label,sublabel:unit==='users'?'명':unit==='requests'?'건':unit==='percent'?'%':unit==='USD'?'달러':unit,data:createMetricSeries(120+i,period,{base,trend,weekly:weekly/base,noise:noise/base,start:START}),goodDirection,format:(v:number)=>unit==='USD'?formatMoney(v):unit==='percent'?v.toFixed(2)+'%':v.toLocaleString('ko-KR',{maximumFractionDigits:0})})),[period]);
 return <ChartShowcase title="운영 현황" subtitle="고정 시드로 생성한 지표 7개의 추이입니다. 스파크라인을 탐색하면 모든 행이 같은 날짜로 맞춰지며, 행을 선택하면 해당 추이를 강조합니다." reference="spark-table" density={density} controls={<div style={group}><label style={ctl}>조회 기간 <select value={period} onChange={e=>{setPeriod(+e.target.value);setIndex(null);}}><option value="30">30일</option><option value="90">90일</option><option value="180">180일</option></select></label><button type="button" onClick={()=>setIndex(null)}>최신값으로 돌아가기</button><span aria-live="polite">{index===null?"최신값":fmtDate(rows[0].data[index]?.t||START)}</span></div>}><SparkTable rows={rows} tone={tone} rowHeight={54} periodLabel={`가상 데이터 · 최근 ${period}일`} index={index} onIndexChange={setIndex} selectedId={selected} onSelectedChange={setSelected} ariaLabel="공통 시점 탐색 기능이 있는 가상 운영 지표"/></ChartShowcase>;
}
function OrderBookDemo({tone,density}:{tone:'light'|'dark';density:Density}){
 const feed=useMemo(()=>createOrderBookFeed({seed:77,mid:184.5,tick:.05,depth:80,rate:5,burstEvery:0}),[]);
 const initial=useMemo(()=>feed.snapshot(),[feed]);
 const bookRef=useRef<OrderBookHandle>(null);
 const [snapshot,setSnapshot]=useState(initial),[paused,setPaused]=useState(false),[layout,setLayout]=useState<OrderBookLayout>('auto'),[linked,setLinked]=useState(true),[selected,setSelected]=useState("호가를 선택하면 누적 수량을 확인할 수 있습니다.");
 // Subscribe only after mount; dispose cancels the demo timer when this block exits.
 useEffect(()=>{const dispose=feed.subscribe(message=>{bookRef.current?.apply(message);setSnapshot(feed.snapshot());});return dispose;},[feed]);
 useEffect(()=>feed.setPaused(paused),[feed,paused]);
 const step=()=>{bookRef.current?.applyDelta(feed.step());setSnapshot(feed.snapshot());};
 return <ChartShowcase title="호가와 시장 구조" subtitle="고정 시드로 재현한 가상 거래소입니다. 호가창, 잔량 비율, 누적 깊이 차트가 같은 주문 데이터를 사용합니다." reference="order-book" density={density} controls={<div style={group}><label style={ctl}>배치 <select value={layout} onChange={e=>setLayout(e.target.value as OrderBookLayout)}><option value="auto">반응형</option><option value="side-by-side">나란히</option><option value="stacked">위아래</option></select></label><label style={ctl}><input type="checkbox" checked={paused} onChange={e=>setPaused(e.target.checked)}/>시뮬레이션 일시 정지</label><button type="button" onClick={step} disabled={!paused}>한 단계 진행</button><label style={ctl}><input type="checkbox" checked={linked} onChange={e=>setLinked(e.target.checked)}/>깊이 차트 연결</label></div>}><div style={{display:'grid',gridTemplateColumns:linked?'repeat(auto-fit,minmax(min(100%,340px),1fr))':'1fr',gap:24,alignItems:'start'}}><div style={{minHeight:420}}><OrderBook ref={bookRef} bids={initial.bids} asks={initial.asks} lastPrice={initial.last} tone={tone} rows={12} rowHeight={25} layout={layout} tickSizes={[.05,.1,.25,.5,1]} defaultTickSize={.05} precision={2} priceLabel="가격 (달러)" sizeLabel="수량" showImbalance flash onPriceSelect={level=>setSelected(`${level.side==='bid'?'매수':'매도'} ${formatMoney(level.price)} · ${level.size.toLocaleString('ko-KR')}개 · 누적 수량 ${level.total.toLocaleString('ko-KR')}`)} ariaLabel="고정 시드 가상 호가창. 키보드로 가격을 선택할 수 있습니다."/></div>{linked&&<DepthChart bids={snapshot.bids} asks={snapshot.asks} height={420} tone={tone} ariaLabel="가상 호가창과 연결된 누적 깊이"/>}</div><p aria-live="polite" style={{fontSize:12,marginTop:12}}>{selected}</p></ChartShowcase>;
}

 const depthFactory=useMemo(()=>createOrderBook(43,{mid:184.5,tick:.05,levels:100}),[]),[depth,setDepth]=useState(()=>depthFactory.snapshot());

 if(variant==='drawdown')return <ChartShowcase title="낙폭과 회복" subtitle="3년간의 가상 자산 가치와 비교 지수를 낙폭 패널과 함께 보여줍니다. 큰 하락 구간을 선택해 최고점, 최저점, 회복일을 확인하세요." reference="drawdown-chart" density={density} controls={<div style={group}><label style={ctl}><input type="checkbox" checked={benchmark} onChange={e=>setBenchmark(e.target.checked)}/>비교 지수</label><label style={ctl}>낙폭 순위 <select value={topN} onChange={e=>setTopN(+e.target.value)}><option value="3">상위 3개</option><option value="4">상위 4개</option><option value="6">상위 6개</option></select></label></div>}><DrawdownChart data={EQUITY.equity} benchmark={benchmark?EQUITY.benchmark:undefined} benchmarkLabel="가상 비교 지수" topN={topN} defaultSelected={0} height={440} tone={tone} ariaLabel="연결된 낙폭 패널과 하락 구간 순위를 갖춘 가상 자산 가치 차트"/></ChartShowcase>;
